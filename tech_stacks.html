Vapi Voice AI Integration

// Vapi Voice AI Integration Example
const vapi = require('@vapi-ai/web');

const assistant = new vapi.Assistant({
  apiKey: 'your-vapi-api-key',
  assistantId: 'your-assistant-id',
  config: {
    tts: {
      provider: 'elevenlabs',
      voiceId: 'voice-id'
    },
    llm: {
      provider: 'openai',
      model: 'gpt-4'
    }
  }
});

// Start the conversation
assistant.start();

// Handle assistant events
assistant.on('message', (message) => {
  console.log('Assistant:', message.text);
});

ElevenLabs Text-to-Speech

# ElevenLabs TTS Integration
import requests

def generate_speech(text, voice_id="your-voice-id"):
    url = f"https://api.elevenlabs.io/v1/text-to-speech/{voice_id}"
    
    headers = {
        "Accept": "audio/mpeg",
        "Content-Type": "application/json",
        "xi-api-key": "your-elevenlabs-api-key"
    }
    
    data = {
        "text": text,
        "model_id": "eleven_monolingual_v1",
        "voice_settings": {
            "stability": 0.5,
            "similarity_boost": 0.5
        }
    }
    
    response = requests.post(url, json=data, headers=headers)
    
    if response.status_code == 200:
        with open("speech.mp3", "wb") as f:
            f.write(response.content)
        return "speech.mp3"
    else:
        return None

AWS Lambda Integration

// AWS Lambda Function for OpenAI Integration
const AWS = require('aws-sdk');
const { Configuration, OpenAIApi } = require('openai');

const secretsManager = new AWS.SecretsManager();

exports.handler = async (event) => {
    // Get OpenAI API key from AWS Secrets Manager
    const secretData = await secretsManager.getSecretValue({
        SecretId: 'openai-api-key'
    }).promise();
    
    const openaiApiKey = JSON.parse(secretData.SecretString).apiKey;
    
    // Configure OpenAI
    const configuration = new Configuration({
        apiKey: openaiApiKey,
    });
    const openai = new OpenAIApi(configuration);
    
    // Process the request
    try {
        const { prompt } = JSON.parse(event.body);
        
        const response = await openai.createCompletion({
            model: "gpt-4",
            prompt: prompt,
            max_tokens: 1000,
            temperature: 0.7,
        });
        
        return {
            statusCode: 200,
            body: JSON.stringify({
                result: response.data.choices[0].text.trim()
            }),
        };
    } catch (error) {
        return {
            statusCode: 500,
            body: JSON.stringify({ error: error.message }),
        };
    }
};
